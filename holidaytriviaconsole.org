server {
    listen 80;
    server_name holidaytriviaconsole.org www.holidaytriviaconsole.org;
    return 301 https://$host$request_uri;
}

server {
   listen 443 ssl;
   server_name holidaytriviaconsole.org www.holidaytriviaconsole.org;

   ssl_certificate /etc/cloudflare/holidaytriviaconsole.org/fullchain.pem;
   ssl_certificate_key /etc/cloudflare/holidaytriviaconsole.org/privatekey.pem;

   location /api {
        proxy_pass http://localhost:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }

    location /ws {
        proxy_pass http://localhost:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }

    location / {
        proxy_pass http://localhost:8000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }
}
